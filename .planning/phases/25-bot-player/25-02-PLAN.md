  <files>packages/server/src/game/BotPlayer.ts, packages/server/tests/game/BotPlayer.test.ts</files>
  <behavior>
    The BotPlayer class provides a `selectMove(gameState, playerId)` static method that returns a `BotMove` describing what action the bot should take.

    BotMove type:
    ```typescript
    export type BotMove =
      | { type: 'play'; cardIndices: number[] }      // Play cards from hand or face-up
      | { type: 'pickup' }                           // Pick up the discard pile
      | { type: 'face-down'; faceDownIndex: number } // Play a face-down card blind
    ```

    Cases:
    1. First turn → find all lowest-rank cards (skip 2s), return their indices as a single play
    2. Hand cards available + valid cards exist → group by rank, play all copies of lowest valid rank
    3. Hand cards available + no valid cards → pickup
    4. Face-up cards (hand empty, draw pile empty) + valid exist → play lowest valid face-up group
    5. Face-up cards + no valid → pickup
    6. Face-down cards only → pick random index (0 to faceDown.length-1)
    7. 7-constraint active → only consider cards with rank <= 7 as valid
    8. Special cards (2, 8, 10) → always valid, but prefer playing normal low cards first unless no normal cards are valid
    9. Empty discard pile → play lowest rank (anything valid)

    Strategy summary: "Play lowest valid rank, group same-rank cards together." This is a simple but competent strategy.
  </behavior>
  <implementation>
    Create `packages/server/src/game/BotPlayer.ts`:

    ```typescript
    import type { GameState, PlayerGameState, Card, PlaySource } from '@shit-head/shared';
    import { canPlayOnPile, getRankValue, RANK_ORDER } from '@shit-head/shared';
    import { GameEngine } from './GameEngine';

    export type BotMove =
      | { type: 'play'; cardIndices: number[] }
      | { type: 'pickup' }
      | { type: 'face-down'; faceDownIndex: number };

    export class BotPlayer {
      /**
       * Selects the best move for a bot given the current game state.
       * Strategy: play lowest valid rank, group same-rank cards together.
       */
      static selectMove(state: GameState, playerId: string): BotMove {
        const player = state.players.find(p => p.playerId === playerId);
        if (!player) throw new Error(`Bot player ${playerId} not found`);

        const playSource = GameEngine.determinePlaySource(player, state.drawPile.length === 0);

        if (playSource === 'face-down') {
          // Random face-down card
          const idx = Math.floor(Math.random() * player.faceDown.length);
          return { type: 'face-down', faceDownIndex: idx };
        }

        const cards = playSource === 'hand' ? player.hand : player.faceUp;

        if (state.firstTurn && playSource === 'hand') {
          return this.selectFirstTurnMove(cards);
        }

        return this.selectBestPlay(cards, state.discardPile);
      }

      private static selectFirstTurnMove(hand: Card[]): BotMove {
        // Must play all copies of lowest rank (skip 2s)
        const scanOrder = RANK_ORDER.slice(1); // skip '2'
        for (const rank of scanOrder) {
          const indices = hand
            .map((c, i) => (c.kind === 'standard' && c.rank === rank) ? i : -1)
            .filter(i => i !== -1);
          if (indices.length > 0) {
            return { type: 'play', cardIndices: indices };
          }
        }
        // Fallback: play index 0
        return { type: 'play', cardIndices: [0] };
      }

      private static selectBestPlay(cards: Card[], discardPile: Card[]): BotMove {
        // Group valid cards by rank value, find lowest valid group
        const validByRank = new Map<number, number[]>(); // rankValue -> indices

        for (let i = 0; i < cards.length; i++) {
          if (canPlayOnPile(cards[i], discardPile)) {
            const rv = getRankValue(cards[i]);
            const group = validByRank.get(rv) || [];
            group.push(i);
            validByRank.set(rv, group);
          }
        }

        if (validByRank.size === 0) {
          return { type: 'pickup' };
        }

        // Find lowest valid rank value
        let lowestRv = Infinity;
        for (const rv of validByRank.keys()) {
          if (rv < lowestRv) lowestRv = rv;
        }

        return { type: 'play', cardIndices: validByRank.get(lowestRv)! };
      }
    }
    ```
