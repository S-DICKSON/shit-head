Create two GitHub Actions workflows per user requirements:

1. **.github/workflows/ci.yml** (runs on PRs):
   ```yaml
   name: CI

   on:
     pull_request:
       branches: [main]

   jobs:
     quality:
       name: Lint, Type Check, Test
       runs-on: ubuntu-latest
       steps:
         - uses: actions/checkout@v4

         - name: Setup Bun
           uses: oven-sh/setup-bun@v2

         - name: Install dependencies
           run: bun install --frozen-lockfile

         - name: Type check (client)
           run: cd packages/client && bunx vue-tsc --noEmit

         - name: Type check (server)
           run: bunx tsc --noEmit --project packages/server/tsconfig.json

         - name: Test (client)
           run: cd packages/client && bunx vitest run

         - name: Test (server)
           run: cd packages/server && bunx vitest run

         - name: Build (client)
           run: cd packages/client && bunx vite build
   ```

   Notes:
   - Uses oven-sh/setup-bun@v2 (latest stable action)
   - Runs type checking, tests, and build verification
   - No lint step yet (ESLint not fully configured in Phase 1 — add a placeholder comment)
   - Each step runs independently so failures are specific

2. **.github/workflows/deploy.yml** (runs on push to main):
   ```yaml
   name: Deploy

   on:
     push:
       branches: [main]

   jobs:
     deploy-server:
       name: Deploy Server to Render
       runs-on: ubuntu-latest
       steps:
         - uses: actions/checkout@v4

         - name: Deploy Server
           uses: JorgeLNJunior/render-deploy@v1.5.0
           with:
             service_id: ${{ secrets.RENDER_SERVER_SERVICE_ID }}
             api_key: ${{ secrets.RENDER_API_KEY }}
             wait_deploy: true

     deploy-client:
       name: Deploy Client to Render
       runs-on: ubuntu-latest
       steps:
         - uses: actions/checkout@v4

         - name: Deploy Client
           uses: JorgeLNJunior/render-deploy@v1.5.0
           with:
             service_id: ${{ secrets.RENDER_CLIENT_SERVICE_ID }}
             api_key: ${{ secrets.RENDER_API_KEY }}
             wait_deploy: true
   ```

   Notes:
   - Server and client deploy in parallel (separate jobs)
   - Secrets (RENDER_API_KEY, RENDER_SERVER_SERVICE_ID, RENDER_CLIENT_SERVICE_ID) must be configured in GitHub repo settings — this is documented but not done here (user_setup handles this)
   - wait_deploy: true means the job waits for Render to finish deploying before marking success
   - Render auto-deploys from GitHub can also be configured as backup, but this gives explicit CI/CD control

Ensure .github/workflows/ directory structure is created.

Create OpenTofu (Terraform-compatible) configuration for Render services:

1. **infra/opentofu/main.tf**:
   ```hcl
   terraform {
     required_version = ">= 1.6.0"

     required_providers {
       render = {
         source  = "render-oss/render"
         version = "~> 1.0"
       }
     }
   }

   provider "render" {
     api_key = var.render_api_key
   }

   # Static site for Vue client
   resource "render_web_service" "client" {
     name    = "${var.project_name}-client"
     plan    = "free"
     region  = var.region

     runtime_source = {
       native_runtime = {
         auto_deploy   = true
         branch        = "main"
         build_command  = "cd packages/client && bun install && bunx vite build"
         repo_url      = var.repo_url
         runtime       = "static"
         publish_path  = "packages/client/dist"
       }
     }

     env_vars = {
       "NODE_ENV" = { value = "production" }
     }
   }

   # Docker web service for Bun server
   resource "render_web_service" "server" {
     name    = "${var.project_name}-server"
     plan    = "free"
     region  = var.region

     runtime_source = {
       docker = {
         auto_deploy      = true
         branch           = "main"
         dockerfile_path  = "./packages/server/Dockerfile"
         context          = "."
         repo_url         = var.repo_url
       }
     }

     env_vars = {
       "NODE_ENV" = { value = "production" }
       "PORT"     = { value = "3000" }
     }
   }
   ```

   Note: The exact Render Terraform provider resource syntax may vary. Check the provider docs during execution and adjust field names as needed. The research references render-oss/render provider. If the exact resource attributes differ, adapt to match the provider's schema.
